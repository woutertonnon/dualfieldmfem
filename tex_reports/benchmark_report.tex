\section*{Introduction}

\[
	U_c=\mathrm{ND}_k,\quad Q_c=\mathrm{CG}_k,\quad
	U_d=\mathrm{RT}_{k-1},\quad W_d=\mathrm{ND}_k,\quad Q_d=\mathrm{DG}_{k-1}.
\]
\[
	\omega_d := w_2,\qquad \omega_c := \nabla\times u_1,\qquad
	\alpha := \frac{1}{\Delta t}\ \text{(main step)},\ \alpha := \frac{2}{\Delta t}\ \text{(start-up)}.
\]
\[
	\mathcal F_h^i=\text{interior faces},\quad h_F=\sqrt{|J_F|},\quad
	[\![\cdot]\!]=\text{two-sided jump},\quad \langle\cdot,\cdot\rangle_F=\int_F \cdot\,\cdot\,dS.
\]

\[
	\begin{bmatrix}
		A_c(\omega_d) & B_c      \\
		B_c^\top      & C_\delta
	\end{bmatrix}
	\begin{bmatrix}u_1 \\ p_1\end{bmatrix}
	=
	\begin{bmatrix}F_c \\ G_c + D_\delta\end{bmatrix},
	\qquad
	\begin{bmatrix}
		A_d(\omega_c) & B_d^\top & C_d^\top \\
		B_d           & D_d      & 0        \\
		C_d           & 0        & 0
	\end{bmatrix}
	\begin{bmatrix}u_2 \\ w_2 \\ p_2\end{bmatrix}
	=
	\begin{bmatrix}F_d \\ 0 \\ 0\end{bmatrix}.
\]

\begin{align}
	a_c(u,v;\omega_d,\beta)
	         & := \alpha (u,v)_\Omega
	+ \nu(\nabla\times u,\nabla\times v)_\Omega
	+ (\omega_d\times u,v)_\Omega                                                                       \\
	         & \quad + \sum_{F\in\mathcal F_h^i}\frac{\sigma_c\nu}{h_F}
	\langle [\![u]\!],[\![v]\!]\rangle_F
	+ \sum_{F\in\mathcal F_h^i}\gamma_c\nu h_F
	\langle [\![\nabla\times u]\!],[\![\nabla\times v]\!]\rangle_F                                      \\
	         & \quad + \sum_{F\in\mathcal F_h^i}\eta_u\,|\beta\cdot n_F|\,
	\langle [\![u]\!],[\![v]\!]\rangle_F
	+ \sum_{F\subset\partial\Omega}\nu\langle n\times(\nabla\times u),v\rangle_F                        \\
	         & \quad + \sum_{F\subset\partial\Omega}\nu\theta\langle u,n\times(\nabla\times v)\rangle_F
	+ \sum_{F\subset\partial\Omega}\frac{\nu C_w^c}{h_F}\langle n\times u,n\times v\rangle_F,
	\\[0.2em]
	b_c(p,v) & := (\nabla p,v)_\Omega,
	\qquad b_c^\top(u,q):=(u,\nabla q)_\Omega,
	\qquad c_\delta(p,q):=-\delta(\nabla p,\nabla q)_\Omega,
	\\[0.2em]
	F_c(v)
	         & := \alpha(u_1^{\mathrm{prev}},v)_\Omega + (f,v)_\Omega
	+ \sum_{F\subset\Gamma_D}\nu\theta\langle u_D,n\times(\nabla\times v)\rangle_F
	+ \sum_{F\subset\Gamma_D}\frac{\nu C_w^c}{h_F}\langle n\times u_D,n\times v\rangle_F,
	\\[0.2em]
	G_c(q)   & :=\langle u_D\cdot n,q\rangle_{\Gamma_D},
	\qquad D_\delta(q):=\delta\,(f+\alpha u_1^{\mathrm{prev}},\nabla q)_\Omega.
\end{align}

\begin{align}
	a_d(u,v;\omega_c)
	              & := \alpha (u,v)_\Omega + (\omega_c\times u,v)_\Omega
	+ \sum_{F\in\mathcal F_h^i}\frac{\sigma_d\nu}{h_F}
	\langle [\![u]\!],[\![v]\!]\rangle_F                                                                                   \\
	              & \quad + \sum_{F\in\mathcal F_h^i}\gamma_d\nu h_F
	\langle [\![\nabla\cdot u]\!],[\![\nabla\cdot v]\!]\rangle_F
	+ \sum_{F\subset\partial\Omega}\frac{C_w^d}{h_F}\langle n\times u,n\times v\rangle_F,
	\\[0.2em]
	b_d(u,\eta)   & := (u,\nabla\times\eta)_\Omega + \langle n\times u,\eta\rangle_{\partial\Omega}
	= (\nabla\times u,\eta)_\Omega,
	\\[0.2em]
	b_d^\top(w,v) & := \nu(\nabla\times w,v)_\Omega,
	\qquad d_d(w,\eta):=-(w,\eta)_\Omega,
	\\[0.2em]
	c_d(u,r)      & :=-(\nabla\cdot u,r)_\Omega,
	\qquad c_d^\top(p,v):=-(p,\nabla\cdot v)_\Omega,
	\\[0.2em]
	F_d(v)
	              & := \alpha(u_2^{\mathrm{prev}},v)_\Omega + (f,v)_\Omega
	+ \sum_{F\subset\Gamma_D}\frac{C_w^d}{h_F}\langle n\times u_D,n\times v\rangle_F,
	\\[0.2em]
	              & \text{with }u_2\cdot n\text{ imposed strongly on RT essential DOFs (all faces, or lid-marker subset).}
\end{align}

\[
	(\sigma_c,\gamma_c,\eta_u,C_w^c,\theta,\delta)
	=(\texttt{hcurl\_sigma},\texttt{hcurl\_gamma},\texttt{hcurl\_upwind},\texttt{hcurl\_Cw},\texttt{hcurl\_theta},\texttt{hcurl\_delta}),
\]
\[
	(\sigma_d,\gamma_d,C_w^d)
	=(\texttt{hdiv\_sigma},\texttt{hdiv\_gamma},\texttt{hdiv\_Cw}).
\]

\section{TaylorGreenCombinedConvergence}

\[
	\nu=0.01,\qquad k=1,\qquad T=0.125,\qquad
	\Delta t(r,o)=\frac{0.5}{8\,2^{ro}}.
\]
\[
	E(t):=e^{-0.02t},\qquad E_2(t):=e^{-0.04t}.
\]
\begin{align}
	u(x,t)
	       & =\begin{bmatrix}
		          E(t)\sin(x_0)\cos(x_1)  \\
		          -E(t)\cos(x_0)\sin(x_1) \\
		          0
	          \end{bmatrix},
	\qquad
	p(x,t)=0,
	\\[0.2em]
	\omega(x,t)=\nabla\times u
	       & =\begin{bmatrix}
		          0 \\0\\2E(t)\sin(x_0)\sin(x_1)
	          \end{bmatrix},
	\\[0.2em]
	f(x,t)
	       & =\partial_t u-u\times(\nabla\times u)-\nu\Delta u+\nabla p
	=\begin{bmatrix}
		 2E_2(t)\sin(x_0)\cos(x_0)\sin^2(x_1) \\
		 2E_2(t)\sin^2(x_0)\sin(x_1)\cos(x_1) \\
		 0
	 \end{bmatrix},
	\\[0.2em]
	u_0(x) & =u(x,0),\qquad \omega_0(x)=\omega(x,0),\qquad u_D(x,t)=u(x,t)\ \text{on }\partial\Omega.
\end{align}
